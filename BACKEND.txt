// Register admin (untuk pertama kali)
const registerAdmin = async (req, res) => {
  try {
    const { username, password, email } = req.body;

    // Check if admin already exists
    const existingAdmin = await User.findOne({ role: 'admin' });
    if (existingAdmin) {
      return res.status(400).json({
        success: false,
        message: 'Admin already registered'
      });
    }

    // Create new admin
    const hashedPassword = await hashPassword(password);
    
    const admin = new User({
      username,
      password: hashedPassword,
      email,
      role: 'admin'
    });

    await admin.save();

    // Generate token
    const token = generateToken(admin._id, admin.username, admin.role);

    res.status(201).json({
      success: true,
      message: 'Admin registered successfully',
      token,
      user: {
        id: admin._id,
        username: admin.username,
        role: admin.role,
        email: admin.email
      }
    });

  } catch (error) {
    console.error('Register error:', error);
    res.status(500).json({
      success: false,
      message: 'Server error',
      error: error.message
    });
  }
};

const login = async (req, res) => {
  try {
    const { username, password, role } = req.body;

    // Find user
    const user = await User.findOne({ username });
    if (!user) {
      return res.status(401).json({
        success: false,
        message: 'Invalid credentials'
      });
    }

    // Check if user is active
    if (!user.isActive) {
      return res.status(403).json({
        success: false,
        message: 'Account is disabled'
      });
    }

    // Check role (admin harus login sebagai admin, user sebagai user)
    if (role && user.role !== role) {
      return res.status(401).json({
        success: false,
        message: 'Invalid role selection'
      });
    }

    // Verify password
    const isPasswordValid = await comparePassword(password, user.password);
    if (!isPasswordValid) {
      return res.status(401).json({
        success: false,
        message: 'Invalid credentials'
      });
    }

    // Update last login
    user.lastLogin = new Date();
    await user.save();

    // Generate token
    const token = generateToken(user._id, user.username, user.role);

    res.json({
      success: true,
      message: 'Login successful',
      token,
      user: {
        id: user._id,
        username: user.username,
        role: user.role,
        email: user.email
      }
    });

  } catch (error) {
    console.error('Login error:', error);
    res.status(500).json({
      success: false,
      message: 'Server error',
      error: error.message
    });
  }
};

const requestEmailChangeCode = async (req, res) => {
  try {
    const { newEmail } = req.body;

    if (!newEmail || !newEmail.trim()) {
      return res.status(400).json({
        success: false,
        message: 'Email baru wajib diisi'
      });
    }

    const email = newEmail.trim().toLowerCase();
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

    if (!emailRegex.test(email)) {
      return res.status(400).json({
        success: false,
        message: 'Format email tidak valid'
      });
    }

    const user = await User.findById(req.user.id);

    if (!user) {
      return res.status(404).json({
        success: false,
        message: 'User not found'
      });
    }

    const currentEmail = (user.email || '').trim().toLowerCase();

    if (currentEmail === email) {
      return res.status(400).json({
        success: false,
        message: 'Email baru sama dengan email saat ini'
      });
    }

    const existingUser = await User.findOne({
      email,
      _id: { $ne: user._id }
    });

    if (existingUser) {
      return res.status(400).json({
        success: false,
        message: 'Email sudah digunakan akun lain'
      });
    }

    const code = generateResetCode();
    const expiry = new Date(Date.now() + 15 * 60 * 1000);

    user.pendingEmail = email;
    user.emailChangeCode = code;
    user.emailChangeCodeExpiry = expiry;
    await user.save();

    try {
      await sendEmailChangeCodeEmail(email, code, user.username);
      console.log(`✅ Email change code sent to ${email}`);
    } catch (emailError) {
      console.error('❌ Email send error:', emailError);

      // FIXED: rollback — hapus semua field verifikasi agar user bisa
      // request ulang tanpa status "pendingEmail" yang menggantung.
      try {
        user.pendingEmail = undefined;
        user.emailChangeCode = undefined;
        user.emailChangeCodeExpiry = undefined;
        await user.save();
        console.log('↩️  Email change state rolled back');
      } catch (rollbackError) {
        console.error('❌ Rollback email change failed:', rollbackError.message);
      }

      return res.status(500).json({
        success: false,
        message: 'Gagal mengirim email. Periksa koneksi server atau coba lagi.'
      });

const updateProfile = async (req, res) => {
  try {
    const { username, email, emailChangeCode } = req.body;

    if (!username || !username.trim()) {
      return res.status(400).json({
        success: false,
        message: 'Username is required'
      });

    if (!user) {
      return res.status(404).json({
        success: false,
        message: 'User not found'
      });
    }

    const currentEmail = (user.email || '').trim().toLowerCase();
    const requestedEmail = email !== undefined
      ? (email ? email.trim().toLowerCase() : '')
      : currentEmail;

    const emailChanged = requestedEmail !== currentEmail;

    // Validasi format email jika diisi
    if (requestedEmail) {
      const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
      if (!emailRegex.test(requestedEmail)) {
        return res.status(400).json({
          success: false,
          message: 'Invalid email format'
        });
      }
    }

    // Jika email berubah, wajib verifikasi kode
    if (emailChanged) {
      if (requestedEmail === '') {
        return res.status(400).json({
          success: false,
          message: 'Email baru wajib diisi'
        });
      }

      if (!emailChangeCode || !emailChangeCode.trim()) {
        return res.status(400).json({
          success: false,
          message: 'Kode verifikasi email wajib diisi'
        });
      }

      if (!user.pendingEmail || user.pendingEmail !== requestedEmail) {
        return res.status(400).json({
          success: false,
          message: 'Silakan minta kode verifikasi untuk email baru terlebih dahulu'
        });
      }

      if (!user.emailChangeCode || !user.emailChangeCodeExpiry) {
        return res.status(400).json({
          success: false,
          message: 'Kode verifikasi tidak ditemukan. Silakan minta kode baru.'
        });
      }

      if (new Date() > new Date(user.emailChangeCodeExpiry)) {
        return res.status(400).json({
          success: false,
          message: 'Kode verifikasi sudah kedaluwarsa. Silakan minta kode baru.'
        });
      }

      if (user.emailChangeCode !== emailChangeCode.trim()) {
        return res.status(400).json({
          success: false,
          message: 'Kode verifikasi salah'
        });
      }
    }

    // Cek username unik
    if (username.trim() !== user.username) {
      const existing = await User.findOne({
        username: username.trim(),
        _id: { $ne: user._id }
      });

      if (existing) {
        return res.status(400).json({
          success: false,
          message: 'Username already taken'
        });
      }
    }

    user.username = username.trim();

    if (emailChanged) {
      user.email = requestedEmail;

      // Bersihkan data verifikasi
      user.pendingEmail = undefined;
      user.emailChangeCode = undefined;
      user.emailChangeCodeExpiry = undefined;
    } else if (email !== undefined) {
      user.email = requestedEmail;
    }

    await user.save();

    res.json({
      success: true,
      message: 'Profile updated successfully',
      user: {
        id: user._id,
        username: user.username,
        email: user.email,
        role: user.role

const changePassword = async (req, res) => {
  try {
    const { oldPassword, newPassword } = req.body;

    if (!oldPassword || !newPassword) {
      return res.status(400).json({
        success: false,
        message: 'Old and new password are required'
      });
    }

    if (newPassword.length < 8) {
      return res.status(400).json({
        success: false,
        message: 'Password must be at least 8 characters'
      });
    }

    if (!/^(?=.*[a-zA-Z])(?=.*\d)/.test(newPassword)) {
      return res.status(400).json({
        success: false,
        message: 'Password must contain both letters and numbers'
      });
    }

    const user = await User.findById(req.user.id);
    if (!user) {
      return res.status(404).json({
        success: false,
        message: 'User not found'
      });
    }

    const isValid = await comparePassword(oldPassword, user.password);
    if (!isValid) {
      return res.status(401).json({
        success: false,
        message: 'Old password is incorrect'
      });
    }

    if (oldPassword === newPassword) {
      return res.status(400).json({
        success: false,
        message: 'New password must be different from old password'
      });
    }

    user.password = await hashPassword(newPassword);
    await user.save();

    res.json({
      success: true,
      message: 'Password changed successfully'
    });
  } catch (error) {
    console.error('Change password error:', error);
    res.status(500).json({
      success: false,
      message: 'Server error',
      error: error.message
    });
  }
};

const { verifyToken } = require('../utils/auth');

const authMiddleware = (req, res, next) => {
  // Get token from header
  const token = req.header('Authorization')?.replace('Bearer ', '');

  if (!token) {
    return res.status(401).json({ 
      success: false, 
      message: 'Access denied. No token provided.' 
    });
  }

  try {
    // Verify token
    const decoded = verifyToken(token);
    
    if (!decoded) {
      return res.status(401).json({ 
        success: false, 
        message: 'Invalid or expired token.' 
      });
    }

    // Add user info to request
    req.user = decoded;
    next();
  } catch (error) {
    return res.status(401).json({ 
      success: false, 
      message: 'Invalid token.' 
    });
  }
};

const adminMiddleware = (req, res, next) => {
  if (req.user.role !== 'admin') {
    return res.status(403).json({ 
      success: false, 
      message: 'Access denied. Admin only.' 
    });
  }
  next();
};

module.exports = { authMiddleware, adminMiddleware };

// Helper function untuk convert string ke ObjectId

  try {
    return new mongoose.Types.ObjectId(id);
  } catch (error) {
    return null;
  }
};

// ============================================================
// Helper: Format tanggal WIB
// ============================================================
const WIB_OFFSET_MS = 7 * 60 * 60 * 1000;

/** Kembalikan tanggal hari ini (WIB) dalam format YYYY-MM-DD */
function getTodayWIBDateStr() {
  const now = new Date();
  const wib = new Date(now.getTime() + WIB_OFFSET_MS);
  return wib.toISOString().split('T')[0];
}

/** Format timestamp menjadi "HH:MM" dalam zona WIB */
function formatWIBTimeLabel(timestamp) {
  const wibDate = new Date(new Date(timestamp).getTime() + WIB_OFFSET_MS);
  const hh = wibDate.getUTCHours().toString().padStart(2, '0');
  const mm = wibDate.getUTCMinutes().toString().padStart(2, '0');
  return `${hh}:${mm}`;
}

// ============================================================
// GET sensor data by date untuk grafik history (WIB aware)
// ============================================================
// FIXED (2026-11):
//   - Query rentang tanggal menggunakan zona WIB (+07:00)
//     sehingga pukul 00:00 WIB s/d 23:59 WIB terambil penuh.
//   - $hour & $minute diberi timezone "+07:00" agar label
//     chart menunjukkan jam WIB, bukan UTC.
//   - Fallback raw data juga diformat ke WIB (bukan UTC).
//   - ✅ BARU: timestamp hasil group di-rebuild via $dateFromParts
//     sehingga SELALU jatuh tepat di menit :00 atau :30 (detik = 0).
//     Ini mencegah label aneh seperti 16:46 di chart frontend.
//
// Pipeline:
//   1. $match      — filter roomId + rentang tanggal WIB
//   2. $addFields  — hitung slot interval (WIB)
//   3. $sort       — urutkan ascending (WAJIB untuk $first)
//   4. $group      — per (hour, interval), ambil $first
//   5. $sort       — urutkan hasil akhir kronologis
//   6. $project    — rebuild timestamp ke slot (detik = 0),
//                    format timeLabel "HH:MM" (WIB)
// ============================================================
router.get('/:roomId/history', async (req, res) => {
  try {
    const { roomId } = req.params;
    const { date, interval = 30 } = req.query;

    if (!mongoose.Types.ObjectId.isValid(roomId)) {
      return res.status(400).json({
        success: false,
        message: 'Invalid room ID format'
      });
    }

    // ✅ FIX: Waktu WIB (UTC+7)
    let startDate, endDate;
    if (date) {
      startDate = new Date(`${date}T00:00:00.000+07:00`);
      endDate = new Date(`${date}T23:59:59.999+07:00`);
    } else {
      const wibDate = getTodayWIBDateStr();
      startDate = new Date(`${wibDate}T00:00:00.000+07:00`);
      endDate = new Date(`${wibDate}T23:59:59.999+07:00`);
    }

    const intervalNum = parseInt(interval, 10);
    if (isNaN(intervalNum) || intervalNum <= 0) {
      return res.status(400).json({
        success: false,
        message: 'Parameter interval tidak valid',
      });
    }

    console.log(
      `📊 Fetching history data for room ${roomId} from ${startDate.toISOString()} to ${endDate.toISOString()} (interval: ${intervalNum}m)`
    );

    const aggregationPipeline = [
      {
        $match: {
          roomId: toObjectId(roomId),
          timestamp: { $gte: startDate, $lte: endDate }
        }
      },
      {
        $addFields: {
          // ✅ FIX: timezone "+07:00" agar $minute ikut WIB
          timeInterval: {
            $subtract: [
              { $minute: { date: '$timestamp', timezone: '+07:00' } },
              {
                $mod: [
                  { $minute: { date: '$timestamp', timezone: '+07:00' } },
                  intervalNum
                ]
              }
            ]
          }
        }
      },
      {
        $sort: { timestamp: 1 }
      },
      {
        $group: {
          _id: {
            hour: { $hour: { date: '$timestamp', timezone: '+07:00' } },
            interval: '$timeInterval'
          },
          timestamp: { $first: '$timestamp' },
          aqi: { $first: '$aqi' },
          pm25: { $first: '$pm25' },
          pm10: { $first: '$pm10' },
          co: { $first: '$co' },
          so2: { $first: '$so2' },
          o3: { $first: '$o3' },
          no2: { $first: '$no2' },
          temperature: { $first: '$temperature' },
          humidity: { $first: '$humidity' },
          category: { $first: '$category' },
          rawData: { $push: '$$ROOT' }
        }
      },
      {
        $sort: { timestamp: 1 }
      },
      {
        $project: {
          _id: 0,
          // ✅ FIX: rebuild timestamp dari slot (hour + interval, detik = 0)
          // sehingga selalu jatuh tepat di menit :00 atau :30.
          // Frontend history_chart.dart menggunakan field ini untuk
          // menampilkan label jam di sumbu-X.
          timestamp: {
            $dateFromParts: {
              year:   { $year:   { date: '$timestamp', timezone: '+07:00' } },
              month:  { $month:  { date: '$timestamp', timezone: '+07:00' } },
              day:    { $dayOfMonth: { date: '$timestamp', timezone: '+07:00' } },
              hour:   '$_id.hour',
              minute: '$_id.interval',
              second: 0,
              millisecond: 0,
              timezone: '+07:00',
            },
          },
          hour: '$_id.hour',
          interval: '$_id.interval',
          timeLabel: {
            $concat: [
              {
                $cond: {
                  if: { $lt: ['$_id.hour', 10] },
                  then: { $concat: ['0', { $toString: '$_id.hour' }] },
                  else: { $toString: '$_id.hour' },
                },
              },
              ':',
              {
                $cond: {
                  if: { $lt: ['$_id.interval', 10] },
                  then: { $concat: ['0', { $toString: '$_id.interval' }] },
                  else: { $toString: '$_id.interval' },
                },
              },
            ],
          },
          aqi: 1,
          pm25: 1,
          pm10: 1,
          co: 1,
          so2: 1,
          o3: 1,
          no2: 1,
          temperature: 1,
          humidity: 1,
          category: 1,
        }
      }
    ];

    const aggregatedData = await SensorData.aggregate(aggregationPipeline);

    // Fallback: jika tidak ada data agregasi, ambil data mentah
    if (aggregatedData.length === 0) {
      const rawData = await SensorData.find({
        roomId: toObjectId(roomId),
        timestamp: { $gte: startDate, $lte: endDate }
      }).sort({ timestamp: 1 });

      return res.json({
        success: true,
        // ✅ FIX: bulatkan timestamp ke slot 30 menit terdekat (detik = 0)
        // dan format timeLabel ke WIB (bukan getHours() yang UTC).
        data: rawData.map((item) => {
          const obj = item.toObject();

          const wibMs = new Date(item.timestamp).getTime() + WIB_OFFSET_MS;
          const wibDate = new Date(wibMs);
          const roundedMinute = wibDate.getUTCMinutes() < 30 ? 0 : 30;
          const roundedWib = new Date(Date.UTC(
            wibDate.getUTCFullYear(),
            wibDate.getUTCMonth(),
            wibDate.getUTCDate(),
            wibDate.getUTCHours(),
            roundedMinute,
            0,
            0,
          ));

          obj.timestamp = new Date(roundedWib.getTime() - WIB_OFFSET_MS);
          obj.timeLabel = formatWIBTimeLabel(obj.timestamp);
          return obj;
        }),
        aggregated: false,
        count: rawData.length,
        message: 'Raw data loaded'
      });

      data: aggregatedData,
      aggregated: true,
      count: aggregatedData.length,
      startDate,
      endDate,
      message: 'Aggregated history data loaded'
    });
  } catch (error) {
    console.error('Error fetching history data:', error);
    res.status(500).json({
      success: false,
      message: 'Error fetching history data',
      error: error.message
    });
  }

router.get('/:roomId/monthly', async (req, res) => {
  try {
    const { roomId } = req.params;
    const year  = parseInt(req.query.year, 10);
    const month = parseInt(req.query.month, 10);

    if (!mongoose.Types.ObjectId.isValid(roomId)) {
      return res.status(400).json({
        success: false,
        message: 'Invalid room ID format',
      });
    }

    if (isNaN(year) || year < 2000 || year > 2100) {
      return res.status(400).json({
        success: false,
        message: 'Parameter year tidak valid',
      });
    }

    if (isNaN(month) || month < 1 || month > 12) {
      return res.status(400).json({
        success: false,
        message: 'Parameter month tidak valid (1-12)',
      });
    }

    // ✅ FIX: gunakan zona WIB agar cut-off tanggal akurat
    const startOfMonth = new Date(
      `${year}-${String(month).padStart(2, '0')}-01T00:00:00.000+07:00`
    );
    // Hari terakhir bulan: tanggal 0 dari bulan berikutnya
    const lastDay = new Date(year, month, 0).getDate();
    const endOfMonth = new Date(
      `${year}-${String(month).padStart(2, '0')}-${String(lastDay).padStart(2, '0')}T23:59:59.999+07:00`
    );

    console.log(
      `📊 Monthly trend → room ${roomId} | ${year}-${String(month).padStart(2, '0')} | ` +
      `${startOfMonth.toISOString()} → ${endOfMonth.toISOString()}`
    );

    const allParams = [
      'aqi', 'pm25', 'pm10', 'co', 'so2', 'o3', 'no2',
      'temperature', 'humidity',
    ];

    // Accumulator tahap 1: per tanggal
    const dailyAcc = {};
    allParams.forEach((p) => {
      dailyAcc[`daily_max_${p}`] = { $max: `$${p}` };
      dailyAcc[`daily_avg_${p}`] = { $avg: `$${p}` };
      dailyAcc[`daily_min_${p}`] = { $min: `$${p}` };
    });

    // Accumulator tahap 2: per hari-dalam-minggu
    const monthlyAcc = { weekCount: { $sum: 1 } };
    allParams.forEach((p) => {
      monthlyAcc[`max_${p}`] = { $max: `$daily_max_${p}` };
      monthlyAcc[`avg_${p}`] = { $avg: `$daily_avg_${p}` };
      monthlyAcc[`min_${p}`] = { $min: `$daily_min_${p}` };
    });

    const pipeline = [
      // 1. Filter: room + rentang bulan + BUKAN anomali + BUKAN error (-1)
      {
        $match: {
          roomId: toObjectId(roomId),
          timestamp: { $gte: startOfMonth, $lte: endOfMonth },
          isAnomaly: { $ne: true },
          aqi: { $gte: 0 },
        },
      },
      // 2. Hitung index hari dalam minggu (Senin=1 .. Minggu=7) — WIB
      {
        $addFields: {
          dayOfWeekMonSun: {
            $add: [
              {
                $mod: [
                  {
                    $add: [
                      { $dayOfWeek: { date: '$timestamp', timezone: '+07:00' } },
                      5,
                    ],
                  },
                  7,
                ],
              },
              1,
            ],
          },
        },
      },
      // 3. Group per tanggal (hari kalender WIB)
      {
        $group: {
          _id: {
            day: { $dayOfMonth: { date: '$timestamp', timezone: '+07:00' } },
            dow: '$dayOfWeekMonSun',
          },
          ...dailyAcc,
        },
      },
      // 4. Group per hari-dalam-minggu
      {
        $group: {
          _id: '$_id.dow',
          ...monthlyAcc,
        },
      },
      // 5. Urutkan Senin..Minggu
      { $sort: { _id: 1 } },
    ];

    const rawResult = await SensorData.aggregate(pipeline);

    const dayNames = [
      'Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu', 'Minggu',
    ];

    const round1 = (v) =>
      typeof v === 'number' && !Number.isNaN(v)
        ? Math.round(v * 10) / 10
        : null;

    const days = [];
    for (let i = 1; i <= 7; i++) {
      const found = rawResult.find((r) => r._id === i);

      const values = {};
      allParams.forEach((p) => {
        values[p] = {
          max: found ? round1(found[`max_${p}`]) : null,
          avg: found ? round1(found[`avg_${p}`]) : null,
          min: found ? round1(found[`min_${p}`]) : null,
        };
      });

      const hasData = values.aqi.max !== null;

      days.push({
        day: i,
        dayName: dayNames[i - 1],
        hasData,
        weekCount: found?.weekCount ?? 0,
        values,
      });
    }

    res.json({
      success: true,
      year,
      month,
      roomId,
      startDate: startOfMonth,
      endDate: endOfMonth,
      days,
      message: 'Monthly trend data loaded',
    });
  } catch (error) {
    console.error('Error fetching monthly trend:', error);
    res.status(500).json({
      success: false,
      message: 'Error fetching monthly trend',
      error: error.message,
    });
  }
});

const BORDER_TOP = '┌────────────────────────────────────────────────────────────────────────────────────';
const BORDER_BOTTOM = '└────────────────────────────────────────────────────────────────────────────────────';

class IoTService {
  constructor() {
    this.isRunning = false;
    this.task = null;
    this.io = null;
    this.deviceLastData = new Map();
    this.pollingInterval = 30000;
  }

  start(io) {
    if (this.isRunning) {
      console.log('⚠️ IoT Service is already running');
      return;
    }

    this.io = io;
    console.log(`🚀 Starting IoT Service (polling every ${this.pollingInterval/1000}s)...`);

    // 🔥 PERBAIKAN: IoT berjalan di detik ke-15 dan ke-45.
    // Simulasi berjalan di detik ke-0 dan ke-30.
    // Ini mencegah keduanya bertabrakan (race condition) di detik yang sama.
    this.task = cron.schedule('15,45 * * * * *', async () => {
      await this.pollAllIoTDevices();
    });

    this.isRunning = true;
    console.log('✅ IoT Service started');
    this.pollAllIoTDevices();
  }

  stop() {
    if (this.task) {
      this.task.stop();
      this.isRunning = false;
      console.log('⏹️ IoT Service stopped');
    }

  async pollAllIoTDevices() {
  const { runExclusive } = require('../utils/serviceLock');

  return runExclusive(async () => {
    try {
      const devices = await IoTDevice.find({});

      if (devices.length === 0) {
        console.log('ℹ️ No IoT devices to poll');
        return;
      }

      console.log(`🔄 Polling ${devices.length} IoT devices...`);

      // ✅ Poll paralel, TAPI tanpa logging border di dalam pollDevice.
      // pollDevice hanya mengembalikan result + error message.
      const results = await Promise.allSettled(
        devices.map(async (device) => {
          try {
            const res = await this.pollDevice(device);
            return { ...res, deviceName: device.name };
          } catch (error) {
            return { success: false, deviceName: device.name, error: error.message };
          }
        })
      );

      // ✅ Log error SERIAL setelah semua polling selesai —
      // tidak ada lagi border berselang-seling.
      let successfulPolls = 0;
        for (const r of results) {
          const val = r.status === 'fulfilled' ? r.value : { success: false, deviceName: 'unknown', error: r.reason?.message };
          if (val.success) {
            successfulPolls++;
          } else {
            console.error(BORDER_TOP);
            console.error(`❌ Error polling device ${val.deviceName}: ${val.error}`);
            console.error(BORDER_BOTTOM);
          }
        }

      console.log(`\n✅ IoT polling completed: ${successfulPolls}/${devices.length} successful`);
      console.log(`🕒 Updated at ${new Date().toLocaleTimeString()}`);
      console.log('\n=====================================================================================\n');
    } catch (error) {
      console.error('❌ Error in pollAllIoTDevices:', error.message);
    }
  });
}

  async pollDevice(device) {
  try {
    console.log(`📡 Polling device: ${device.name} (${device.apiEndpoint})`);
    const response = await axios.get(device.apiEndpoint, { timeout: 5000 });

    if (response.data?.success === true) {
        const iotData = response.data.data;

        if (!this.isValidIoTData(iotData)) {
          throw new Error('Invalid IoT data structure');
        }

        const lastData = this.deviceLastData.get(device._id.toString());
        const hasChanged = this._hasDataChanged(lastData, iotData);

        this.deviceLastData.set(device._id.toString(), iotData);

        device.status = 'online';
        device.lastUpdate = new Date();
        await device.save();

        // ✅ SELALU panggil updateRoomFromIoT — biarkan method itu
        // yang memutuskan perlu save/broadcast atau tidak,
        // TAPI tetap mencetak ringkasan AQI setiap polling.
        const rooms = await Room.find({
          iotDeviceId: device._id.toString()
        });

        for (const room of rooms) {
          await this.updateRoomFromIoT(room, iotData, device.name);
        }

        return { success: true, changed: hasChanged };
      }

      return {
        success: false,
        deviceName: device.name,
        error: 'Device did not return success',
      };
      } catch (error) {
        device.status = 'offline';
        await device.save().catch(() => {});
        return {
          success: false,
          deviceName: device.name,
          error: error.message,
        };
      }
    }

      })

    const updates = results
      .filter((r) => r.status === 'fulfilled' && r.value)
      .map((r) => r.value);

    if (updates.length > 0) {
      const bulkOps = updates.map((u) => ({
        updateOne: {

        await IoTDevice.bulkWrite(bulkOps);

  isValidIoTData(data) {
    if (!data || typeof data !== 'object') return false;

    const knownFields = [
      'aqi', 'pm25', 'pm10', 'co', 'so2', 'o3', 'no2',
      'temperature', 'humidity'

  _hasDataChanged(oldData, newData) {
    if (!oldData || !newData) return true;

    const changed = (oldV, newV) => {
      if (oldV == null && newV == null) return false;
      if (oldV == null || newV == null) return true;
      return Math.abs(Number(newV) - Number(oldV)) > 0;
    };

    return (
      changed(oldData.aqi, newData.aqi) ||
      changed(oldData.pm25, newData.pm25) ||
      changed(oldData.pm10, newData.pm10) ||
      changed(oldData.co, newData.co) ||
      changed(oldData.so2, newData.so2) ||
      changed(oldData.o3, newData.o3) ||
      changed(oldData.no2, newData.no2) ||
      changed(oldData.temperature, newData.temperature) ||
      changed(oldData.humidity, newData.humidity)
    );

/**
 * ============================================================================
 * UNIT TEST — src/utils/aqiCalculator.js
 * ============================================================================
 *
 * Menguji 4 fungsi publik dari aqiCalculator.js:
 *   1. calculateISPUFromPM25(conc)       → ISPU 0-500 (dari PM2.5)
 *   2. calculateISPUFromPM10(conc)       → ISPU 0-500 (dari PM10)
 *   3. calculateAQI(pm25, pm10, co)      → { aqi, dominant, ispuPM25, ... }
 *   4. getAQICategory(aqi)               → { category, label, color }
 *
 * Standar acuan: PP No. 22 Tahun 2021 (Indonesia).
 *
 * ----------------------------------------------------------------------------
 * RUMUS INTERPOLASI LINEAR
 * ----------------------------------------------------------------------------
 *   ISPU = ((ISPU_high − ISPU_low) / (C_high − C_low)) × (C − C_low) + ISPU_low
 *
 * ----------------------------------------------------------------------------
 * KATEGORI ISPU (PP 22/2021)
 * ----------------------------------------------------------------------------
 *   0   –  50   : Baik
 *   51  – 100   : Sedang
 *   101 – 200   : Tidak Sehat
 *   201 – 300   : Sangat Tidak Sehat
 *   301 – 500   : Berbahaya
 *
 * ----------------------------------------------------------------------------
 * POLA PENGUJIAN TIAP FUNGSI
 * ----------------------------------------------------------------------------
 *   A. NILAI DI TENGAH SEGMEN  → verifikasi rumus interpolasi benar
 *   B. NILAI TEPAT DI BOUNDARY → pastikan tidak "bocor" ke segmen sebelah
 *   C. PERPINDAHAN SEGMEN      → pastikan titik "naik kelas" benar (+1 unit)
 *   D. NILAI EKSTREM / INVALID → pastikan clamp & sanitasi input bekerja
 *
 * ----------------------------------------------------------------------------
 * STRUKTUR FILE
 * ----------------------------------------------------------------------------
 *   BAGIAN 1 — calculateISPUFromPM25
 *   BAGIAN 2 — calculateISPUFromPM10
 *   BAGIAN 3 — calculateAQI  (MAX dari ketiga parameter)
 *   BAGIAN 4 — getAQICategory (batas antar kategori)

//   ─────         ───────────────────

describe('calculateISPUFromPM25 — konversi PM2.5 (µg/m³) → ISPU', () => {

  // ── A. Nilai di tengah segmen (verifikasi rumus interpolasi) ─────────────

  test('PM2.5 = 0 → ISPU 0 (titik awal, tanpa interpolasi)', () => {
    assert.strictEqual(calculateISPUFromPM25(0), 0);
  });

  test('PM2.5 = 27.5 → ISPU 25 (tengah segmen 0–55)', () => {
    // (50 − 0)/(55 − 0) × (27.5 − 0) + 0 = 25
    assert.strictEqual(calculateISPUFromPM25(27.5), 25);
  });

  test('PM2.5 = 100 → ISPU 74 (tengah segmen 56–150)', () => {

    //   = 22.94 + 51
    //   ≈ 73.93  → dibulatkan 74
    assert.strictEqual(calculateISPUFromPM25(100), 74);

  // ── B. Nilai tepat di boundary (tidak boleh "bocor" ke segmen sebelah) ──

  test('PM2.5 = 55 → ISPU 50 (batas atas segmen Baik, BUKAN 51)', () => {
    assert.strictEqual(calculateISPUFromPM25(55), 50);

  test('PM2.5 = 150 → ISPU 100 (batas atas segmen Sedang)', () => {

//
// ATURAN TIE (nilai ISPU sama di beberapa parameter):

//     • CO tie dengan parameter lain   → CO menang

  test('PM10 dominan: PM2.5=20, PM10=60 → AQI 55, dominant "pm10"', () => {

    // PM10 : (49/99) × (60−51) + 51        = 55.45 → 55

    // CO   : 0                              = 0
    // MAX(74, 30, 0) = 74 → PM2.5 dominan
    const r = calculateAQI(100, 30);
    assert.strictEqual(r.aqi, 74);
    assert.strictEqual(r.dominant, 'pm25');
  });

  test('CO dominan: PM2.5=10, PM10=20, CO=20000 → AQI 234, dominant "co"', () => {
    // PM2.5: (50/55) × 10                  ≈ 9
    // PM10 : (50/50) × 20                  = 20
    // CO   : segmen 15001–30000 µg/m³      → (99/14999) × (20000−15001) + 201 ≈ 234
    // MAX(9, 20, 234) = 234 → CO dominan
    const r = calculateAQI(10, 20, 20000);
    assert.strictEqual(r.aqi, 234);
    assert.strictEqual(r.dominant, 'co');

  });

  test('Tie PM2.5 vs PM10, CO rendah → dominant "pm25"', () => {
    // PM2.5 = 30 → ISPU (50/55) × 30 ≈ 27
    // PM10  = 27 → ISPU (50/50) × 27 = 27
    // CO    =  0 → ISPU = 0 (tidak ikut tie)
    // Tie PM2.5 vs PM10 → PM2.5 menang (dicek lebih dulu)
    const r = calculateAQI(30, 27, 0);

    assert.ok(r.ispuPM25 > r.ispuCO);
    assert.strictEqual(r.dominant, 'pm25');
  });

  test('Tie 3 arah (semua ISPU = 0) → dominant "co"', () => {
    // PM2.5 = 0, PM10 = 0, CO = 0 → semua ISPU = 0
    // Tie 3 arah → CO menang (dicek lebih dulu)

    assert.strictEqual(r.aqi, 0);
    assert.strictEqual(r.ispuPM25, 0);
    assert.strictEqual(r.ispuPM10, 0);
    assert.strictEqual(r.ispuCO, 0);
    assert.strictEqual(r.dominant, 'co');
  });

  // ── Skenario 3: konsistensi category & label ─────────────────────────────

  test('Kategori mengikuti nilai MAX: AQI 55 → category "sedang", label "Sedang"', () => {

    assert.strictEqual(r.aqi, 0);
    assert.strictEqual(r.category, 'baik');
  });

  test('Input sangat tinggi (600, 600) → AQI 500 (clamped), category "berbahaya"', () => {
    const r = calculateAQI(600, 600);
    assert.strictEqual(r.aqi, 500);
    assert.strictEqual(r.category, 'berbahaya');

const isValidNumber = (v) =>
  typeof v === 'number' && Number.isFinite(v) && !Number.isNaN(v);

function calculateISPUFromConcentration(concentration, breakpoints) {
  if (!isValidNumber(concentration) || concentration < 0) return null;

  for (const bp of breakpoints) {
    if (concentration <= bp.concHigh) {
      const concRange = bp.concHigh - bp.concLow;
      if (concRange === 0) return bp.ispuHigh;
      const ispu =
        ((bp.ispuHigh - bp.ispuLow) / concRange) *
          (concentration - bp.concLow) +
        bp.ispuLow;
      return Math.round(Math.max(0, Math.min(500, ispu)));
    }
  }
  return 500;
}

/**
 * AQI = MAX dari parameter yang tersedia.
 * Semua parameter nullable. Bila SEMUA null → aqi: null, category: 'no_data'.
 * Tie-breaker: CO → SO2 → O3 → NO2 → PM2.5 → PM10
 */
function calculateAQI(pm25 = null, pm10 = null, co = null, so2 = null, o3 = null, no2 = null) {
  const hasPM25 = isValidNumber(pm25);
  const hasPM10 = isValidNumber(pm10);
  const hasCO   = isValidNumber(co);
  const hasSO2  = isValidNumber(so2);
  const hasO3   = isValidNumber(o3);
  const hasNO2  = isValidNumber(no2);

  if (!hasPM25 && !hasPM10 && !hasCO && !hasSO2 && !hasO3 && !hasNO2) {
    const cat = getAQICategory(null);
    return {
      aqi: null,
      ispuPM25: null, ispuPM10: null, ispuCO: null,
      ispuSO2: null, ispuO3: null, ispuNO2: null,
      dominant: null,
      category: cat.category,
      color: cat.color,
      label: cat.label,
    };
  }

  const ispuPM25 = hasPM25 ? calculateISPUFromPM25(pm25) : null;
  const ispuPM10 = hasPM10 ? calculateISPUFromPM10(pm10) : null;
  const ispuCO   = hasCO   ? calculateISPUFromCO(co)     : null;
  const ispuSO2  = hasSO2  ? calculateISPUFromSO2(so2)   : null;
  const ispuO3   = hasO3   ? calculateISPUFromO3(o3)     : null;
  const ispuNO2  = hasNO2  ? calculateISPUFromNO2(no2)   : null;

  const ordered = [
    ...(hasCO   ? [{ name: 'co',   value: ispuCO   }] : []),
    ...(hasSO2  ? [{ name: 'so2',  value: ispuSO2  }] : []),
    ...(hasO3   ? [{ name: 'o3',   value: ispuO3   }] : []),
    ...(hasNO2  ? [{ name: 'no2',  value: ispuNO2  }] : []),
    ...(hasPM25 ? [{ name: 'pm25', value: ispuPM25 }] : []),
    ...(hasPM10 ? [{ name: 'pm10', value: ispuPM10 }] : []),
  ];

  const aqi = Math.max(...ordered.map((c) => c.value));
  const dominant = ordered.find((c) => c.value === aqi)?.name || null;

  const cat = getAQICategory(aqi);

  return {
    aqi,
    ispuPM25, ispuPM10, ispuCO, ispuSO2, ispuO3, ispuNO2,
    dominant,
    category: cat.category,
    color: cat.color,
    label: cat.label,
  };
}

function getAQICategory(aqi) {
  if (aqi == null || Number.isNaN(aqi)) {
    return { category: 'no_data', color: '#BDBDBD', label: 'Tidak Ada Data' };
  }
  if (aqi <= 50)  return { category: 'baik',               color: '#00E400', label: 'Baik' };
  if (aqi <= 100) return { category: 'sedang',             color: '#0000FF', label: 'Sedang' };
  if (aqi <= 200) return { category: 'tidak_sehat',        color: '#FFFF00', label: 'Tidak Sehat' };
  if (aqi <= 300) return { category: 'sangat_tidak_sehat', color: '#FF0000', label: 'Sangat Tidak Sehat' };
  return { category: 'berbahaya', color: '#000000', label: 'Berbahaya' };
}

function aqiToConcentration(aqi, breakpoints) {
  if (!isValidNumber(aqi) || aqi <= 0) return 0;
  const target = Math.min(500, aqi);
  for (const bp of breakpoints) {
    if (target <= bp.ispuHigh) {
      const ispuRange = bp.ispuHigh - bp.ispuLow;
      if (ispuRange === 0) return bp.concLow;
      const ratio = (target - bp.ispuLow) / ispuRange;
      return Math.round(bp.concLow + ratio * (bp.concHigh - bp.concLow));
    }
  }
  return 500;
}

// Hash password
const hashPassword = async (password) => {
  const salt = await bcrypt.genSalt(10);
  return await bcrypt.hash(password, salt);
};

// Compare password
const comparePassword = async (password, hashedPassword) => {
  return await bcrypt.compare(password, hashedPassword);
};

  } catch (error) {
    return null;
  }

function getTimeSlot(hour) {
  if (hour >= 6  && hour < 8)  return '06-08';
  if (hour >= 8  && hour < 12) return '08-12';
  if (hour >= 12 && hour < 16) return '12-16';
  if (hour >= 16 && hour < 18) return '16-18';
  if (hour >= 18 && hour < 20) return '18-20';
  return '20-06';
}

function randomInRange(min, max, mean) {
  const mode = Math.max(min, Math.min(max, mean ?? (min + max) / 2));
  const u = Math.random();
  const f = (mode - min) / (max - min);

  if (u < f) {
    return min + Math.sqrt(u * (max - min) * (mode - min));
  }
  return max - Math.sqrt((1 - u) * (max - min) * (max - mode));
}

    const co   = normalData.co * (1.5 + Math.random());
    const so2  = normalData.so2 * multiplier;
    const o3   = normalData.o3 * multiplier;
    const no2  = normalData.no2 * multiplier;

    const aqiResult = calculateAQI(pm25, pm10, co, so2, o3, no2);

    return {
      ...normalData,
      pm25: parseFloat(pm25.toFixed(1)),
      pm10: parseFloat(pm10.toFixed(1)),
      co: Math.round(co),
      so2: parseFloat(so2.toFixed(1)),
      o3: parseFloat(o3.toFixed(1)),
      no2: parseFloat(no2.toFixed(1)),
      aqi: aqiResult.aqi,
      category: aqiResult.category,
      dominant: aqiResult.dominant,
      ispuPM25: aqiResult.ispuPM25,
      ispuPM10: aqiResult.ispuPM10,
      ispuCO: aqiResult.ispuCO,
      ispuSO2: aqiResult.ispuSO2,
      ispuO3: aqiResult.ispuO3,
      ispuNO2: aqiResult.ispuNO2,
      isAnomaly: true,
      anomalyType: 'spike',
    };
  }

  // 3% SENSOR ERROR
  if (roll < 0.08) {
    return {
      ...normalData,
      pm25: -1,
      pm10: -1,
      co: -1,
      so2: -1,
      o3: -1,
      no2: -1,
      temperature: -1,
      humidity: -1,
      aqi: -1,
      category: 'error',
      isAnomaly: true,
      anomalyType: 'error',
    };
  }

  return normalData;
};

/**
 * Service lock — memastikan hanya SATU siklus service
 * (simulation / iot) yang berjalan pada satu waktu.
 *
 * Tujuan: output terminal tidak saling memotong antar service.
 *
 * Cara kerja:
 *   - Setiap pemanggil `runExclusive(fn)` mengantre pada promise chain.
 *   - `fn` dijalankan setelah semua pemanggil sebelumnya selesai.
 *   - Jika `fn` melempar error, promise tetap resolve (tidak merusak chain).
 */

let chain = Promise.resolve();

function runExclusive(fn) {
  const run = chain.then(fn, fn); // jalankan fn baik chain resolve maupun reject
  // chain berikutnya menunggu `run` selesai (success/fail), tanpa
  // meneruskan rejection agar 1 error tidak "meracuni" pemanggil berikutnya.
  chain = run.then(
    () => undefined,
    () => undefined
  );
  return run;
}

module.exports = { runExclusive };

// 11) GRACEFUL SHUTDOWN (penting untuk Render)
// ============================================================
// Render mengirim SIGTERM saat:
//   - deploy versi baru (container lama dimatikan)
//   - manual restart
//   - scale down
// Kita beri waktu 10 detik untuk menutup koneksi DB & socket,
// lalu exit. Tanpa ini, Render akan force-kill setelah 30 detik
// dan mungkin menyisakan koneksi "zombie" di MongoDB Atlas.

let isShuttingDown = false;

const shutdown = async (signal) => {
  if (isShuttingDown) return;
  isShuttingDown = true;

  console.log(`\n⚠️  Received ${signal}. Shutting down gracefully...`);

  // 1) Stop background services
  try {
    simulationService.stop();
    iotService.stop();
    console.log('⏹️  Background services stopped');
  } catch (e) {
    console.error('⚠️  Error stopping services:', e.message);
  }

  // 2) Tutup Socket.IO
  try {
    io.close(() => console.log('🔌 Socket.IO closed'));
  } catch (e) {
    console.error('⚠️  Error closing Socket.IO:', e.message);
  }

  // 3) Tutup HTTP server
  server.close(() => {
    console.log('🛑 HTTP server closed');
  });

  // 4) Disconnect MongoDB
  try {
    await mongoose.disconnect();
    console.log('🔌 MongoDB disconnected');
  } catch (e) {
    console.error('⚠️  Error disconnecting MongoDB:', e.message);
  }

  // 5) Paksa exit setelah 10 detik jika masih menggantung
  setTimeout(() => {
    console.error('⏱️  Forcing exit after 10s');
    process.exit(1);
  }, 10000).unref();

  process.exit(0);
};

process.on('SIGTERM', () => shutdown('SIGTERM'));
process.on('SIGINT', () => shutdown('SIGINT'));

// Tangkap unhandled rejection agar tidak crash diam-diam
process.on('unhandledRejection', (reason) => {
  console.error('❌ Unhandled Rejection:', reason);
});

process.on('uncaughtException', (err) => {
  console.error('❌ Uncaught Exception:', err);
  // Jangan langsung exit — biarkan graceful shutdown jika ada
  shutdown('uncaughtException');
});
